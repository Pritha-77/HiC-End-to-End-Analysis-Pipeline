#!/usr/bin/env python3
"""
Generate restriction enzyme cut sites for all commonly used Hi-C enzymes
"""
 
import re
import os
 
# Common restriction enzymes used in Hi-C experiments
HIC_ENZYMES = {
    'MboI': 'GATC',           # 4-base cutter, most common for high-resolution Hi-C
    'DpnII': 'GATC',          # 4-base cutter, same recognition as MboI
    'HindIII': 'AAGCTT',      # 6-base cutter, good for genome-wide overview
    'NcoI': 'CCATGG',         # 6-base cutter
    'BglII': 'AGATCT',        # 6-base cutter
    'EcoRI': 'GAATTC',        # 6-base cutter
    'BamHI': 'GGATCC',        # 6-base cutter
    'XhoI': 'CTCGAG',         # 6-base cutter
    'SacI': 'GAGCTC',         # 6-base cutter
    'KpnI': 'GGTACC',         # 6-base cutter
    'SalI': 'GTCGAC',         # 6-base cutter
    'SpeI': 'ACTAGT',         # 6-base cutter
    'XbaI': 'TCTAGA',         # 6-base cutter
    'NheI': 'GCTAGC',         # 6-base cutter
    'AluI': 'AGCT',           # 4-base cutter
    'Sau3AI': 'GATC',         # 4-base cutter, same as MboI/DpnII
    'TaqI': 'TCGA',           # 4-base cutter
    'MseI': 'TTAA',           # 4-base cutter
    'CviQI': 'GTAC',          # 4-base cutter
    'HaeIII': 'GGCC'          # 4-base cutter
}
 
def find_restriction_sites(fasta_file, enzyme_name, enzyme_seq, output_file):
    """
    Find all restriction enzyme cut sites in the genome
    """
    sites = []
    current_chr = None
    current_pos = 0
    sequence_buffer = ""
 
    with open(fasta_file, 'r') as f:
        for line_num, line in enumerate(f, 1):
            line = line.strip()
            if line.startswith('>'):
                # Process any remaining sequence in buffer
                if sequence_buffer and current_chr:
                    for match in re.finditer(enzyme_seq, sequence_buffer.upper()):
                        pos = current_pos + match.start() + 1  # 1-based coordinate
                        sites.append(f"{current_chr}\t{pos}")
 
                # New chromosome
                current_chr = line[1:].split()[0]  # Take only chromosome name
                current_pos = 0
                sequence_buffer = ""
            else:
                # DNA sequence - add to buffer
                sequence_buffer += line.upper()
 
                # Process buffer when it gets large enough
                if len(sequence_buffer) > 1000000:  # Process in 1MB chunks
                    # Find sites in current buffer
                    for match in re.finditer(enzyme_seq, sequence_buffer):
                        pos = current_pos + match.start() + 1  # 1-based coordinate
                        sites.append(f"{current_chr}\t{pos}")
 
                    # Keep overlap region for sites spanning chunks
                    overlap = len(enzyme_seq) - 1
                    current_pos += len(sequence_buffer) - overlap
                    sequence_buffer = sequence_buffer[-overlap:]
 
    # Process final buffer
    if sequence_buffer and current_chr:
        for match in re.finditer(enzyme_seq, sequence_buffer):
            pos = current_pos + match.start() + 1  # 1-based coordinate
            sites.append(f"{current_chr}\t{pos}")
 
    # Write restriction sites to file
    with open(output_file, 'w') as f:
        f.write(f"# Restriction sites for {enzyme_name} ({enzyme_seq})\n")
        f.write(f"# Format: chromosome<tab>position\n")
        f.write(f"# Total sites found: {len(sites)}\n")
        for site in sites:
            f.write(site + '\n')
 
    return len(sites)
 
def generate_all_restriction_sites(fasta_file):
    """Generate restriction sites for all Hi-C enzymes"""
 
    if not os.path.exists(fasta_file):
        print(f"Error: Reference genome file not found: {fasta_file}")
        return
 
    total_enzymes = len(HIC_ENZYMES)
    results = {}
 
    for i, (enzyme_name, enzyme_seq) in enumerate(HIC_ENZYMES.items(), 1):
        output_file = f"hg38_{enzyme_name}.txt"
        site_count = find_restriction_sites(fasta_file, enzyme_name, enzyme_seq, output_file)
        results[enzyme_name] = site_count
 
if __name__ == "__main__":
    # Generate sites for the reference genome
    fasta_file = 'hg38.fa'
    generate_all_restriction_sites(fasta_file)
EOF

#!/usr/bin/env bash

# ============================================================
# Hi-C DOWNSTREAM ANALYSIS PIPELINE
# ============================================================
#
# PURPOSE:
#   1. A/B compartment analysis
#   2. Pearson correlation analysis
#   3. Chromatin loop detection
#   4. Loop-anchor extraction
#   5. Gene/promoter/enhancer/CTCF annotation
#   6. Loop-level annotation integration
#   7. Overlap-aware anchor classification
#   8. CTCF association
#   9. Regulatory burden analysis
#  10. Final QC
#
# MAIN TOOLS:
#   FAN-C
#   Juicer Tools
#   BEDTools
#   GENCODE v38
#   ENCODE SCREEN cCREs
#
# GENOME:
#   hg38
#
# ANALYSIS:
#   Chromosome 1
#
# ============================================================


# ============================================================
# SETUP
# ============================================================
## New Enviornment
mamba create -n fanc37 python=3.7 -y
# Activate the FAN-C environment used for Hi-C downstream analysis.
conda activate fanc37

mamba install -c conda-forge -c bioconda fanc -y
fanc --version
pip install Deprecated	
	
mkdir -p /mnt/d/hic/hic_project/downstream/fanc

# Define directories so that paths do not need to be repeatedly written.
HIC="/mnt/d/hic/hic_project/aligned"
FANC="/mnt/d/hic/hic_project/downstream/fanc"
LOOPS="/mnt/d/hic/hic_project/downstream/loops"

# Reference annotation files.
GTF="/mnt/d/hic/hic_references/gencode.v38.annotation.gtf"
CCRE="$LOOPS/GRCh38-cCREs.bed"

# Create output directories if they do not already exist.
mkdir -p "$FANC"
mkdir -p "$LOOPS"


# ============================================================
# A/B COMPARTMENT ANALYSIS
# ============================================================

# ------------------------------------------------------------
# Convert the Juicer .hic contact map into FAN-C format.
#
# Resolution:
#   100 kb
#
# Biological purpose:
#   A/B compartments describe broad-scale chromatin organization.
#
# A compartment:
#   Generally associated with active/open chromatin.
#
# B compartment:
#   Generally associated with inactive/compact chromatin.
# ------------------------------------------------------------

# Convert Juicer .hic → FAN-C at 100-kb resolution
fanc to-fanc \
    "$HIC/inter_30.hic@100000@SCALE" \
    "$FANC/inter_30_100kb.fanc"


# ------------------------------------------------------------
# Calculate A/B compartments using FAN-C.
# ------------------------------------------------------------

fanc compartments \
    "$FANC/inter_30_100kb.fanc" \
    "$FANC/compartments_100kb"

ls -lh /mnt/d/hic/hic_project/downstream/fanc

# ------------------------------------------------------------
# Export the compartment eigenvector values.
#
# Output:
#   AB_compartments_100kb.bed
#
# Note:
#   Eigenvector sign itself is arbitrary. The orientation should
#   be interpreted using biological/reference information rather
#   than assuming positive automatically means A.
# ------------------------------------------------------------

fanc compartments \
    "$FANC/compartments_100kb" \
    -v "$FANC/AB_compartments_100kb.bed"


# ------------------------------------------------------------
# Export continuous compartment domains.
# ------------------------------------------------------------

fanc compartments \
    "$FANC/compartments_100kb" \
    -d "$FANC/AB_domains_100kb.bed"
# ------------------------------------------------------------
# Calculate compartment strength.
#
# Biological purpose:
#   Measures the strength of compartmental segregation.
# ------------------------------------------------------------

fanc compartments \
    "$FANC/inter_30_100kb.fanc" \
    "$FANC/compartments_100kb" \
    --compartment-strength \
    "$FANC/compartment_strength_100kb.txt"

# ------------------------------------------------------------
# Generate compartment enrichment profile.
# ------------------------------------------------------------

fanc compartments \
    "$FANC/inter_30_100kb.fanc" \
    "$FANC/compartments_100kb" \
    -e "$FANC/enrichment_profile_100kb.pdf"



#Final output files#
inter_30_100kb.fanc
compartments_100kb
AB_compartments_100kb.bed
AB_domains_100kb.bed
compartment_strength_100kb.txt
enrichment_profile_100kb.pdf

# ============================================================
# COMPARTMENT VISUALIZATION
# ============================================================

# ------------------------------------------------------------
# Visualize chromosome 1 with the Hi-C contact map and
# A/B compartment track.
# ------------------------------------------------------------

fancplot chr1 \
    -o "$FANC/chr1_AB_compartments.pdf" \
    --plot triangular "$FANC/inter_30_100kb.fanc" \
    --plot bar "$FANC/AB_compartments_100kb.bed"

# ------------------------------------------------------------
# Zoom into the first 50 Mb of chromosome 1.
# ------------------------------------------------------------

fancplot chr1:1-50000000 \
    -o "$FANC/chr1_1_50Mb_AB.pdf" \
    --plot triangular "$FANC/inter_30_100kb.fanc" \
    --plot bar "$FANC/AB_compartments_100kb.bed"

# ------------------------------------------------------------
# Visualize continuous compartment domains.
# Plot compartment domains
# ------------------------------------------------------------

fancplot chr1:1-50000000 \
    -o "$FANC/chr1_AB_domains.pdf" \
    --plot triangular "$FANC/inter_30_100kb.fanc" \
    --plot layer "$FANC/AB_domains_100kb.bed"


# ============================================================
# PEARSON CORRELATION ANALYSIS
# ============================================================

# ------------------------------------------------------------
# Generate a Pearson correlation matrix at 100-kb resolution.
#
# Biological purpose:
#   Pearson correlation transforms the contact matrix into a
#   similarity/correlation matrix that highlights compartmental
#   organization.
#
# Expected pattern:
#   Similar compartment regions → positive correlation
#   Different compartment regions → negative correlation
# ------------------------------------------------------------

~/hic_tools/juicer/scripts/common/juicer_tools pearsons \
    SCALE \
    "$HIC/inter_30.hic" \
    chr1 \
    BP \
    100000 \
    "$HIC/chr1_pearsons_100kb.txt"

# Note: Pearson correlation reveals compartmental organization
#       - Strong positive correlations indicate same compartment type
#       - Negative correlations indicate different compartment types
#       - Creates characteristic plaid pattern in heatmaps
# ============================================================
# CONVERT Hi-C DATA FOR LOOP ANALYSIS
# ============================================================

# ------------------------------------------------------------
# Convert the .hic file to FAN-C format at 25-kb resolution.
#
# Why higher resolution?
#   Chromatin loops are smaller-scale structures than A/B
#   compartments, so a finer resolution is used.
# ------------------------------------------------------------

fanc to-fanc \
    "$HIC/inter_30.hic@25000@SCALE" \
    "$FANC/inter_30.fanc"


# ============================================================
# LOOP DETECTION
# ============================================================

# ------------------------------------------------------------
# Detect chromatin loops on chromosome 1.
#
# Parameters:
#
#   --peak-size 1
#       Defines the local peak size.
#
#   --neighborhood-width 4
#       Defines the surrounding neighborhood used during
#       loop detection.
#
#   --min-distance 2
#       Minimum genomic separation between loop anchors.
#
#   --fdr 0.1
#       False-discovery-rate threshold.
#
#   --observed 5
#       Minimum observed interaction requirement.
#
#   --merge-pixels
#       Merge nearby significant pixels representing the same
#       interaction.
#
#   --merge-distance 20000
#       Maximum distance for merging nearby loop calls.
#
#   --remove-singlets
#       Removes isolated/single-pixel calls.
#
# Output:
#   BEDPE file containing two anchors for every detected loop.
# ------------------------------------------------------------

fanc loops \
    "$FANC/inter_30.fanc" \
    "$LOOPS/inter_30_chr1_loops" \
    --chromosomes chr1 \
    --threads 8 \
    --peak-size 1 \
    --neighborhood-width 4 \
    --min-distance 2 \
    --fdr 0.1 \
    --observed 5 \
    --merge-pixels \
    --merge-distance 20000 \
    --remove-singlets \
    --bedpe "$LOOPS/inter_30_chr1_loops.bedpe" \
    --force-overwrite


# ============================================================
# EXTRACT LOOP ANCHORS
# ============================================================

head -20 \
/mnt/d/hic/hic_project/downstream/loops/inter_30_chr1_loops.bedpe

wc -l \
/mnt/d/hic/hic_project/downstream/loops/inter_30_chr1_loops.bedpe

# ------------------------------------------------------------
# BEDPE contains coordinates for both sides of each loop.
#
# Anchor 1:
#   columns 1-3
#
# Anchor 2:
#   columns 4-6
# ------------------------------------------------------------

awk 'BEGIN{OFS="\t"} {
    print $1,$2,$3
}' "$LOOPS/inter_30_chr1_loops.bedpe" \
> "$LOOPS/chr1_loop_anchor1.bed"


awk 'BEGIN{OFS="\t"} {
    print $4,$5,$6
}' "$LOOPS/inter_30_chr1_loops.bedpe" \
> "$LOOPS/chr1_loop_anchor2.bed"

# Check the anchors
head -10 /mnt/d/hic/hic_project/downstream/loops/chr1_loop_anchor1.bed
head -10 /mnt/d/hic/hic_project/downstream/loops/chr1_loop_anchor2.bed

wc -l /mnt/d/hic/hic_project/downstream/loops/chr1_loop_anchor1.bed
wc -l /mnt/d/hic/hic_project/downstream/loops/chr1_loop_anchor2.bed

# Visualize a detected loop: Take one loop from the BEDPE
head -1 \
/mnt/d/hic/hic_project/downstream/loops/inter_30_chr1_loops.bedpe
# ------------------------------------------------------------
# Remove duplicate coordinates within each anchor set.
# ------------------------------------------------------------

sort -k1,1 -k2,2n -k3,3n \
    "$LOOPS/chr1_loop_anchor1.bed" |
uniq \
> "$LOOPS/chr1_loop_anchor1_unique.bed"


sort -k1,1 -k2,2n -k3,3n \
    "$LOOPS/chr1_loop_anchor2.bed" |
uniq \
> "$LOOPS/chr1_loop_anchor2_unique.bed"


# ------------------------------------------------------------
# Create one non-redundant BED file containing all loop anchors.
#
# This file is used for general annotation/QC.
# Anchor1 and Anchor2 files are retained because later analysis
# must preserve which annotation belongs to which anchor.
# ------------------------------------------------------------

cat \
    "$LOOPS/chr1_loop_anchor1_unique.bed" \
    "$LOOPS/chr1_loop_anchor2_unique.bed" |
sort -k1,1 -k2,2n -k3,3n |
uniq \
> "$LOOPS/chr1_all_loop_anchors.bed"



wc -l /mnt/d/hic/hic_project/downstream/loops/chr1_all_loop_anchors.bed

#Visualize a detected loop (first loop)
head -1 /mnt/d/hic/hic_project/downstream/loops/inter_30_chr1_loops.bedpe

# Visualize a detected loop: visualize a larger surrounding region
fancplot chr1:23000000-24000000 \
    -o /mnt/d/hic/hic_project/downstream/loops/chr1_loop_example.pdf \
    --plot triangular \
    /mnt/d/hic/hic_project/downstream/fanc/inter_30.fanc
	
	
	
# ============================================================
# CREATE LOOP IDs
# ============================================================

# ------------------------------------------------------------
# Add a unique Loop_ID to every detected loop.
#
# This is important because annotation files initially contain
# genomic coordinates but do not retain the original loop ID.
# ------------------------------------------------------------

awk 'BEGIN{OFS="\t"}
{
    print NR,$1,$2,$3,$4,$5,$6
}' "$LOOPS/inter_30_chr1_loops.bedpe" \
> "$LOOPS/chr1_loops_with_ids.bedpe"


# ------------------------------------------------------------
# Create the master loop table.
# ------------------------------------------------------------

awk 'BEGIN{
    OFS="\t";
    print "Loop_ID",
          "Anchor1_chr","Anchor1_start","Anchor1_end",
          "Anchor2_chr","Anchor2_start","Anchor2_end"
}
{
    print $1,$2,$3,$4,$5,$6,$7
}' "$LOOPS/chr1_loops_with_ids.bedpe" \
> "$LOOPS/chr1_loop_master.tsv"


# ============================================================
# GENE ANNOTATION
# ============================================================
## hg38 annotation BED/GTF file is required
## Download:https://ftp.ebi.ac.uk/pub/databases/gencode/Gencode_human/release_38/gencode.v38.annotation.gtf.gz

# Input
LOOPS="/mnt/d/hic/hic_project/downstream/loops"
GTF="/mnt/d/hic/hic_references/gencode.v38.annotation.gtf"

# ------------------------------------------------------------
# Extract chromosome 1 gene records from GENCODE v38.

awk 'BEGIN{OFS="\t"}
$1=="chr1" && $3=="gene" {
    print

# OR #______________Extract gene annotations from GENCODE for the entire genome______________#
awk 'BEGIN{OFS="\t"} 
$3=="gene" {
    print
}' "$GTF" \
> "$LOOPS/gencode_genes.gtf"
# ------------------------------------------------------------
# Convert GENCODE gene records from GTF to BED.
#
# Output fields:
#   chromosome
#   start
#   end
#   gene ID
#   gene name
#   gene type
# ------------------------------------------------------------

head "$LOOPS/chr1_gencode_genes.gtf"

#Convert GENCODE gene GTF → BED
awk 'BEGIN{OFS="\t"}
$1=="chr1" && $3=="gene" {

    attrs=$9

    for(i=10;i<=NF;i++)
        attrs=attrs" "$i

    gene_id="NA"
    gene_name="NA"
    gene_type="NA"

    if (match(attrs,/gene_id "[^"]+"/))
        gene_id=substr(attrs,RSTART+9,RLENGTH-10)

# ------------------------------------------------------------

$1=="chr1" && $3=="gene" {

    attrs=$9

    for(i=10;i<=NF;i++)
        attrs=attrs" "$i

    gene_id="NA"
    gene_name="NA"

    if (match(attrs,/gene_id "[^"]+"/))
        gene_id=substr(attrs,RSTART+9,RLENGTH-10)

    if (match(attrs,/gene_name "[^"]+"/))
        gene_name=substr(attrs,RSTART+11,RLENGTH-12)

        tss=$4
    else if ($7=="-")
        tss=$5
    else
        next

    start=tss-2000

    end=tss+2000

# ------------------------------------------------------------
# The BEDTools intersections contain anchor coordinates but

# The following step maps:
#
#     anchor coordinates → Loop_ID
#
# so every annotation can be linked back to its original loop.

# ------------------------------------------------------------
# Function:
# Add Loop_ID to an annotation table.
#
# Arguments:
#   1 = anchor-loop mapping file
#   2 = BEDTools annotation file
#   3 = output file
# ------------------------------------------------------------

add_loop_id() {

    MAP="$1"
    ANNOT="$2"
    OUT="$3"
    HEADER="$4"

    awk -v HEADER="$HEADER" 'BEGIN{OFS="\t"}

    NR==FNR {

        if (NR>1)
            loop[$2 FS $3 FS $4]=$1

        next
    }

    NR==1 {

        print HEADER

        next
    }

    {

        key=$1 FS $2 FS $3

        if (key in loop)
            print loop[key],$0

    }' "$MAP" "$ANNOT" > "$OUT"
}

# ============================================================
# OVERLAP-AWARE ANCHOR CLASSIFICATION
# ============================================================

# ------------------------------------------------------------
# Classify each anchor independently.
#
# P   = promoter
# E   = enhancer
# P+E = promoter AND enhancer
# O   = no promoter or enhancer annotation
#
# IMPORTANT:
# P+E is preserved as its own category.
# ------------------------------------------------------------

awk -F'\t' 'BEGIN{OFS="\t"}

NR==1 {

    print $0,
          "Anchor1_type",
          "Anchor2_type",
          "Anchor_pair_class"

    next
}

{

    # --------------------------------------------------------
    # Anchor 1
    # --------------------------------------------------------

    A1P=($10>0 || $12>0)
    A1E=($14>0)

    if(A1P && A1E)
        A1="P+E"
    else if(A1P)
        A1="P"
    else if(A1E)
        A1="E"
    else
        A1="O"


    # --------------------------------------------------------
    # Anchor 2
    # --------------------------------------------------------

    A2P=($11>0 || $13>0)
    A2E=($15>0)

    if(A2P && A2E)
        A2="P+E"
    else if(A2P)
        A2="P"
    else if(A2E)
        A2="E"
    else
        A2="O"


    # --------------------------------------------------------
    # OVERLAP-AWARE ANCHOR-PAIR CLASSIFICATION
    # --------------------------------------------------------

    # P+E -- P+E
    if(A1=="P+E" && A2=="P+E")
        pair="P+E-P+E"

    # P+E -- P
    else if((A1=="P+E" && A2=="P") ||
            (A1=="P" && A2=="P+E"))
        pair="P+E-P"

    # P+E -- E
    else if((A1=="P+E" && A2=="E") ||
            (A1=="E" && A2=="P+E"))
        pair="P+E-E"

    # P+E -- O
    else if((A1=="P+E" && A2=="O") ||
            (A1=="O" && A2=="P+E"))
        pair="P+E-O"

    # P -- P
    else if(A1=="P" && A2=="P")
        pair="PP"

    # P -- E or E -- P
    else if((A1=="P" && A2=="E") ||
            (A1=="E" && A2=="P"))
        pair="PE"

    # E -- E
    else if(A1=="E" && A2=="E")
        pair="EE"

    # P -- O or O -- P
    else if((A1=="P" && A2=="O") ||
            (A1=="O" && A2=="P"))
        pair="P-O"

    # E -- O or O -- E
    else if((A1=="E" && A2=="O") ||
            (A1=="O" && A2=="E"))
        pair="E-O"

    # O -- O
    else
        pair="O-O"


    print $0,A1,A2,pair

}

' chr1_loop_integrated_summary.tsv \
> chr1_loop_anchor_pair_classified.tsv

# ------------------------------------------------------------
# FUNCTION TO COLLECT UNIQUE ANNOTATIONS
# ------------------------------------------------------------
function collect(arr, loop,    key, parts, result, first) {

    result=""
    first=1

    for (key in arr) {

        split(key, parts, SUBSEP)

        if (parts[1] == loop) {

            if (!first)
                result=result";"

            result=result parts[2]

            first=0
        }
    }

    if (result == "")
        result="NA"

    return result
}


# ------------------------------------------------------------
# FUNCTION TO COUNT UNIQUE ANNOTATIONS
# ------------------------------------------------------------
function count_annotations(arr, loop,    key, parts, count) {

    count=0

    for (key in arr) {

        if (parts[1] == loop)

    }

    return count
